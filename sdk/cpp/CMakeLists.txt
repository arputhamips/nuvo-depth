cmake_minimum_required(VERSION 3.25)
project(nuvo_depth_client VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(NUVO_CLIENT_WITH_HTTPS "Build verified HTTPS/WebSocket acquisition" ON)
option(NUVO_CLIENT_BUILD_TESTS "Build client contract and transport tests" OFF)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
# The export tool copies this exact dependency closure into a standalone tree.
# Framing has one canonical implementation, shared with the appliance C ABI.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/wire_protocol.cpp")
  set(NUVO_WIRE_SOURCE src/wire_protocol.cpp)
  set(NUVO_WIRE_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/private")
else()
  set(NUVO_WIRE_SOURCE ../../src/wire_protocol.cpp)
  set(NUVO_WIRE_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
endif()
add_library(nuvo_depth_client STATIC src/frame.cpp src/local.cpp src/replay.cpp ${NUVO_WIRE_SOURCE})
add_library(nuvo::client ALIAS nuvo_depth_client)
set_target_properties(nuvo_depth_client PROPERTIES EXPORT_NAME client POSITION_INDEPENDENT_CODE ON)
target_compile_features(nuvo_depth_client PUBLIC cxx_std_20)
target_include_directories(nuvo_depth_client PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE ${NUVO_WIRE_INCLUDES})
target_link_libraries(nuvo_depth_client PRIVATE $<BUILD_INTERFACE:nlohmann_json::nlohmann_json> PUBLIC Threads::Threads)
if(NUVO_CLIENT_WITH_HTTPS)
  find_package(OpenSSL 3 REQUIRED)
  find_package(Boost 1.74 REQUIRED COMPONENTS system)
  find_package(c-ares CONFIG REQUIRED)
  target_sources(nuvo_depth_client PRIVATE src/https.cpp)
  target_link_libraries(nuvo_depth_client PUBLIC OpenSSL::SSL OpenSSL::Crypto Boost::system c-ares::cares)
else()
  target_sources(nuvo_depth_client PRIVATE src/https_disabled.cpp)
endif()
if(NOT MSVC)
  target_compile_options(nuvo_depth_client PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
install(TARGETS nuvo_depth_client EXPORT nuvo_depth_clientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/nuvo DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
  set(NUVO_CLIENT_LICENSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
else()
  set(NUVO_CLIENT_LICENSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
install(FILES "${NUVO_CLIENT_LICENSE_ROOT}/LICENSE" "${NUVO_CLIENT_LICENSE_ROOT}/NOTICE"
  DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/nuvo_depth_client)
install(EXPORT nuvo_depth_clientTargets NAMESPACE nuvo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nuvo_depth_client)
configure_package_config_file(cmake/nuvo_depth_clientConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/nuvo_depth_clientConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nuvo_depth_client)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/nuvo_depth_clientConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/nuvo_depth_clientConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/nuvo_depth_clientConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nuvo_depth_client)
if(NUVO_CLIENT_BUILD_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(nuvo_client_test tests/client_test.cpp)
  target_link_libraries(nuvo_client_test PRIVATE nuvo::client)
  add_test(NAME nuvo_client_contract COMMAND nuvo_client_test)
  add_test(NAME nuvo_client_transports COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/transports.py $<TARGET_FILE:nuvo_client_test>)
  set_tests_properties(nuvo_client_transports PROPERTIES ENVIRONMENT "NUVO_CLIENT_WITH_HTTPS=${NUVO_CLIENT_WITH_HTTPS}")
  add_test(NAME nuvo_client_measurements COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/measurements.py $<TARGET_FILE:nuvo_client_test>)
  add_test(NAME nuvo_client_installed_consumer COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/installed.py ${CMAKE_CURRENT_BINARY_DIR})
endif()
